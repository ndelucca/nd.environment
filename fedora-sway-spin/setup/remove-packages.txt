# Apps que NO queremos en el sistema (vengan del Spin o arrastradas como dependencia
# débil). Las borra 07-remove-unwanted.sh de forma idempotente: solo quita las que
# están instaladas, no falla si ya no están. Una por línea; '#' y vacías se ignoran.
#
# gnome-text-editor: editor de texto nativo de GNOME. No lo usamos (nuestro editor de
# text/plain ya es nvim) y nada lo requiere.
gnome-text-editor

# eog: visor de imágenes viejo (GTK3) de GNOME. Lo reemplazamos por Loupe (GTK4),
# que es el handler de imágenes en 03-apps.sh. Nada lo requiere.
eog

# direnv: estuvo en packages.txt pero nunca se hookeó en bashrc.d (eval "$(direnv hook
# bash)"), así que no hacía nada. Sacado del set; se desinstala para no dejar peso muerto.
direnv

# dunst: daemon de notificaciones de dunst. Nuestro daemon es SwayNotificationCenter
# (lo arranca sway/config), pero dunst quedaba instalado y registrando el mismo nombre
# de D-Bus: el journal repetía en cada arranque
#   Ignoring duplicate name 'org.freedesktop.Notifications' ... org.knopwob.dunst.service
# swaync ganaba la carrera por lanzarse explícito, pero era una condición latente.
# Nada lo requiere.
dunst

# abrt: reporte automático de crashes de Fedora. Deja 4 daemons corriendo (abrtd,
# abrt-oops, abrt-journal-core y abrt-xorg — este último vigila un log de Xorg que en una
# sesión Wayland pura no existe). No lo usamos y en una máquina de 4 GB no se justifica.
# Basta con quitar 'abrt': dnf arrastra los addons, la GUI y libreport como dependencias.
abrt

# low-memory-monitor: avisador de presión de memoria de GNOME, sin usar acá (nada lo
# requiere y no hay GNOME Shell). Su función la cubre systemd-oomd, que sí habilitamos
# en 11-memory.sh.
low-memory-monitor
